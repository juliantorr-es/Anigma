## 🚀 QUICK START

### 1. Basic Usage
```swift
import CathedralModule
import DatabaseCore

// Create database
let db = DatabaseActor(dbPath: "cathedral.db")
try await db.open()

// Create Cathedral
let cathedral = await CathedralModule.createFacade(
    database: db,
    mlService: mlService
)

// Execute ML operation with evidence
let operation = MLOperation(
    type: .embedding,
    sessionId: "session-123",
    agentId: "my-agent",
    parameters: [
        "text": "Hello world",
        "model": "nomic-embed-text-v1.5"
    ]
)

let result = try await cathedral.executeOperation(
    operation: operation,
    requirement: .moderate
)
// ✅ Evidence recorded automatically
```

### 2. Web API Usage
```bash
# Generate embedding
curl -X POST http://localhost:8080/api/v1/ml/embed \
  -H "Content-Type: application/json" \
  -d '{
    "text": "Test document",
    "model": "nomic-embed-text-v1.5"
  }'

# Get compliance report
curl http://localhost:8080/api/v1/evidence/compliance/session-123
```

### 3. Harmonia Integration
```swift
import HarmoniaModule

// Create Cathedral plan compiler
let compiler = CathedralPlanCompiler(cathedral: cathedral)

// Generate evidence-backed plan
let plan = try await compiler.generatePlan(
    request: PlanRequest(
        operationType: "semantic_search",
        sessionContext: ["sessionId": "session-123"],
        parameters: ["query": "contract terms"],
        priority: .high
    )
)
// ✅ Plan approved with evidence
```
